  componentDidMount() {
    this.onResize = () => this.setState({ winW: window.innerWidth });
    window.addEventListener('resize', this.onResize);
    this.onKey = e => {
      const t = e.target && e.target.tagName;
      if (t === 'INPUT' || t === 'TEXTAREA' || t === 'SELECT') return;
      if (this.state.screen !== 'sessao') return;
      if (e.code === 'Space') { e.preventDefault(); if (!this.state.revealed) this.setState({ revealed:true }); }
      if (['1','2','3','4'].includes(e.key) && this.state.revealed) this.rate(+e.key);
    };
    window.addEventListener('keydown', this.onKey);
  }
  componentWillUnmount() {
    window.removeEventListener('resize', this.onResize);
    window.removeEventListener('keydown', this.onKey);
    clearInterval(this.impT); clearTimeout(this.fbT); clearTimeout(this.toastT);
  }

  go = (screen, extra) => {
    this.setState({ screen, modal:null, ...(extra || {}) });
    if (this.mainRef.current) this.mainRef.current.scrollTop = 0;
  };
  toast = (kind, text) => {
    clearTimeout(this.toastT);
    this.setState({ toast:{ kind, text, k: Date.now() } });
    this.toastT = setTimeout(() => this.setState({ toast:null }), kind === 'meta' ? 3800 : 2800);
  };

  startSession = (deckId) => {
    let list = [];
    if (deckId) list = (this.CARDS[deckId] || []).map(c => ({ c, d: deckId }));
    else ['far','ana','fis','pat','cli'].forEach(id => this.CARDS[id].slice(0, 3).forEach(c => list.push({ c, d:id })));
    if (!list.length) return;
    const queue = list.map((x, i) => {
      const dk = this.state.decks.find(d => d.id === x.d);
      return { q:x.c[0], a:x.c[1], deck:dk.nome, deckBg:dk.bg, deckInk:dk.ink, ivl:this.IVL[i % 4], kind:this.KIND[i % 4], again:false };
    });
    const dk = deckId ? this.state.decks.find(d => d.id === deckId) : null;
    this.go('sessao', { queue, pos:0, revealed:false, results:[0,0,0,0], fb:0, sessName: dk ? dk.nome : 'Todos os baralhos', sessStart: Date.now(), metaHitInSession:false });
  };

  rate = (n) => {
    const s = this.state;
    if (!s.revealed || s.fb) return;
    const results = [...s.results]; results[n - 1]++;
    let queue = s.queue;
    if (n === 1) queue = [...queue, { ...queue[s.pos], again:true, kind:'DE NOVO', ivl:['1 min','6 min','10 min','1 dia'] }];
    const answered = s.answered + 1;
    this.setState({ results, queue, fb:n, answered });
    if (answered === s.metaResp && (this.props.celebrate ?? true)) {
      this.setState({ metaHitInSession:true });
      this.toast('meta', 'Meta do dia cumprida. Ofensiva: 13 dias.');
    }
    clearTimeout(this.fbT);
    this.fbT = setTimeout(() => {
      this.setState(st => {
        const pos = st.pos + 1;
        if (pos >= st.queue.length) return { screen:'resultado', fb:0, revealed:false, sessEnd: Date.now() };
        return { pos, fb:0, revealed:false };
      });
    }, n === 4 ? 800 : 620);
  };

  setImp = (patch) => this.setState(s => ({ imp:{ ...s.imp, ...patch } }));
  startImport = (scanned) => {
    clearInterval(this.impT);
    this.setImp({ step:2, prog:0, drag:false, scanned:!!scanned });
    this.impT = setInterval(() => {
      const p = this.state.imp.prog + 2;
      if (this.state.imp.scanned && p >= 34) { clearInterval(this.impT); this.setImp({ step:'erro', prog:34 }); return; }
      if (p >= 100) {
        clearInterval(this.impT);
        const items = this.IMPORT.map((x, i) => ({ id:i + 1, q:x[0], a:x[1], on: i !== 7, dup: i === 7, edit:false }));
        this.setImp({ step:3, prog:100, items, strat:'sep' });
        return;
      }
      this.setImp({ prog:p });
    }, 90);
  };
  setItem = (id, patch) => this.setImp({ items: this.state.imp.items.map(it => it.id === id ? { ...it, ...patch } : it) });

  setCC = (patch) => this.setState(s => ({ cc:{ ...s.cc, ...patch } }));
  saveCard = (again) => {
    const cc = this.state.cc;
    if (!cc.q.trim() || !cc.a.trim()) { this.setCC({ tried:true }); return; }
    const deck = this.state.decks.find(d => d.id === cc.deck);
    this.setState(s => ({ created: s.created + 1, decks: s.decks.map(d => d.id === cc.deck ? { ...d, total:d.total + 1, novos:d.novos + 1 } : d) }));
    this.setCC({ q:'', a:'', tried:false, fromLeech:false });
    this.toast('ok', 'Card salvo em ' + deck.nome + '.');
    if (!again) this.go('baralho', { deckId: cc.deck });
  };

  sessVals() {
    const s = this.state;
    const sCard = s.queue[s.pos] || { q:'', a:'', deck:'', ivl:['','','',''], kind:'' };
    const fbMap = { 1:['#C9485B','volta em 1 min'], 2:['#B8952F','difícil'], 3:['#4F7358','bom ✓'], 4:['#3D6580','fácil ✦'] };
    const fb = fbMap[s.fb];
    const tot = s.results.reduce((a, b) => a + b, 0);
    const secs = Math.max(1, Math.round((s.sessEnd - s.sessStart) / 1000));
    const ret = tot ? Math.round((s.results[2] + s.results[3]) / tot * 100) : 0;
    return {
      sCard, sessName: s.sessName, sPos: Math.min(s.pos + 1, s.queue.length), sTotal: s.queue.length,
      sProgW: (s.queue.length ? (s.pos / s.queue.length * 100) : 0) + '%',
      revealed: s.revealed, notRevealed: !s.revealed,
      reveal: () => this.setState({ revealed:true }),
      rate1: () => this.rate(1), rate2: () => this.rate(2), rate3: () => this.rate(3), rate4: () => this.rate(4),
      ivl1: sCard.ivl[0], ivl2: sCard.ivl[1], ivl3: sCard.ivl[2], ivl4: sCard.ivl[3],
      hasStamp: !!fb, stampC: fb ? fb[0] : '#000', stampT: fb ? fb[1] : '', isFacilFb: s.fb === 4,
      sBorder: fb ? fb[0] : '#E3D9C4', sCardT: s.fb === 3 || s.fb === 4 ? 'translateY(-3px)' : s.fb === 1 ? 'translateX(-3px)' : 'none',
      exitSession: () => { clearTimeout(this.fbT); this.go(s.results.some(x => x) ? 'resultado' : 'estudar', { sessEnd: Date.now(), fb:0 }); },
      resTotal: tot, resRet: ret + '%', resTime: secs < 60 ? secs + 's' : Math.floor(secs / 60) + 'min',
      resDist: [['Errei','#E7A9BD'],['Difícil','#E3CB85'],['Bom','#87BC96'],['Fácil','#A9C1D2']].map((x, i) => ({ l:x[0], c:x[1], n:s.results[i] })),
      resMsg: s.metaHitInSession ? 'E a meta do dia foi batida no meio da sessão. Ofensiva: 13 dias seguidos.' : ret >= 80 ? 'Retenção ótima. Esses cards vão voltar mais espaçados.' : 'Os que você errou voltam logo — é assim que a memória fixa.',
    };
  }

  deckVals(decksV) {
    const s = this.state;
    const dk0 = decksV.find(d => d.id === s.deckId) || decksV[0];
    const dk = { ...dk0, notEmpty: !dk0.empty };
    const states = [['REVISÃO','#F4D9E3','#9C3E66'],['NOVO','#DCEBD9','#3E6B4A'],['APRENDENDO','#F3E3B5','#7A5F14'],['DOMINADO','#E4E8D6','#55643F']];
    const nexts = ['hoje','hoje','amanhã','em 4 dias','em 12 dias','em 1 mês'];
    const q = s.search.trim().toLowerCase();
    const all = (this.CARDS[dk.id] || []).map((c, i) => ({ n: String(i + 1).padStart(2, '0'), q:c[0], a:c[1], state:states[i % 4][0], tb:states[i % 4][1], tc:states[i % 4][2], next:nexts[i % 6] }));
    const dkCards = q ? all.filter(c => (c.q + ' ' + c.a).toLowerCase().includes(q)) : all;
    const cc = s.cc;
    return {
      dk, dkCards, dkShown: dkCards.length, dkNoResults: !!q && !dkCards.length, search: s.search,
      onSearch: e => this.setState({ search: e.target.value }),

      ccPrevA: cc.a.trim() || 'e a resposta, depois de virar.', ccPrevAc: cc.a.trim() ? '#6E6250' : '#B5A88F',
      ccTags: cc.tags.map(t => ({ t, rm: () => this.setCC({ tags: this.state.cc.tags.filter(x => x !== t) }) })),
      onCcQ: e => this.setCC({ q: e.target.value }), onCcA: e => this.setCC({ a: e.target.value }),
      onCcDeck: e => this.setCC({ deck: e.target.value }), onTagIn: e => this.setCC({ tagIn: e.target.value }),
      onTagKey: e => {
        if (e.key !== 'Enter') return; e.preventDefault();
        const t = this.state.cc.tagIn.trim().replace(/^#/, '').toLowerCase();
        if (t && !this.state.cc.tags.includes(t)) this.setCC({ tags:[...this.state.cc.tags, t], tagIn:'' }); else this.setCC({ tagIn:'' });
      },
      saveCard: () => this.saveCard(false), saveCardAgain: () => this.saveCard(true),
    };
  }

  impVals() {
    const s = this.state, imp = s.imp, st = imp.step;
    const stepIdx = st === 'erro' ? 2 : st === 4 ? 4 : st;
    const impSteps = ['ENVIAR','PROCESSANDO','REVISAR'].map((l, i) => {
      const n = i + 1, done = stepIdx > n, act = stepIdx === n, err = st === 'erro' && n === 2;
      return { l: '0' + n + ' — ' + l, sym: done ? '✓' : err ? '!' : '0' + n,
        bg: act ? (err ? '#F6D8D8' : '#FDFBF5') : 'transparent', bd: act ? (err ? '#E7A9BD' : '#D98CAE') : '#E3D9C4',
        cb: done ? '#7FA886' : act ? (err ? '#C9485B' : '#D98CAE') : '#F6EFE0', cc: done || act ? '#FFF' : '#8A7C68', tc: act || done ? '#4A4034' : '#8A7C68' };
    });
    const msgs = ['Lendo seu material','Identificando possíveis perguntas','Montando seus cards'];
    const mi = imp.prog < 34 ? 0 : imp.prog < 70 ? 1 : 2;
    const impChecks = msgs.map((l, i) => ({ l, sym: i < mi ? '✓' : '', col: i <= mi ? '#4A4034' : '#B5A88F',
      bg: i < mi ? '#7FA886' : 'transparent', bd: i < mi ? 'none' : i === mi ? '2px solid #D98CAE' : '1.5px dashed #E3D9C4' }));
    const strat = this.STRATS.find(x => x.id === imp.strat);
    const items = imp.items || [];
    const impItems = items.map((it, i) => ({
      ...it, num: String(i + 1).padStart(2, '0'), view: !it.edit, editL: it.edit ? 'Pronto' : 'Editar',
      op: it.on ? 1 : .55, bd: it.edit ? '#D98CAE' : '#E3D9C4',
      cbBg: it.on ? '#4F7358' : '#FDFBF5', cbBd: it.on ? '#4F7358' : '#D8CCB3', cbSym: it.on ? '✓' : '',
      toggle: () => this.setItem(it.id, { on: !it.on }), onEdit: () => this.setItem(it.id, { edit: !it.edit }),
      onDel: () => this.setImp({ items: this.state.imp.items.filter(x => x.id !== it.id) }),
      onQ: e => this.setItem(it.id, { q: e.target.value }), onA: e => this.setItem(it.id, { a: e.target.value }),
    }));
    const sel = items.filter(x => x.on).length;
    const allOn = items.length && sel === items.length;
    const deck = s.decks.find(d => d.id === imp.deck);
    return {
      imp, impSteps, imp1: st === 1, imp2: st === 2, imp3: st === 3, imp4: st === 4, impErr: st === 'erro',
      dropBd: imp.drag ? '#B03D66' : '#D98CAE', dropBg: imp.drag ? '#F4D9E3' : '#FAEBF0',
      onDragOver: e => { e.preventDefault(); if (!this.state.imp.drag) this.setImp({ drag:true }); },
      onDragLeave: () => this.setImp({ drag:false }),
      onDrop: e => { e.preventDefault(); this.startImport(false); },
      startImport: () => this.startImport(false), startImportScanned: () => this.startImport(true),
      cancelImport: () => { clearInterval(this.impT); this.setImp({ step:1, prog:0 }); },
      restartImport: () => this.setImp({ step:1, prog:0, items:[] }),
      impMsg: msgs[mi], impProgW: imp.prog + '%', impPage: Math.max(1, Math.ceil(imp.prog / 100 * 18)), impChecks,
      impN: items.length, impSel: sel, impConf: strat.c + '%', impLow: strat.c < 60,
      strats: this.STRATS.map(x => ({ id:x.id, lab: x.l + ' · ' + x.c + '%' })),
      onStrat: e => { this.setImp({ strat: e.target.value }); this.toast('ok', 'Pares recalculados.'); },
      onImpDeck: e => this.setImp({ deck: e.target.value }),
      toggleAll: () => this.setImp({ items: items.map(x => ({ ...x, on: !allOn })) }), toggleAllL: allOn ? 'Desmarcar todos' : 'Marcar todos',
      impDeckName: deck ? deck.nome : '', impAdded: imp.added || 0,
      addImport: () => {
        if (!sel) { this.toast('err', 'Marque pelo menos um card.'); return; }
        this.setState(st2 => ({ created: st2.created + sel, decks: st2.decks.map(d => d.id === imp.deck ? { ...d, total: d.total + sel, novos: d.novos + sel } : d) }));
        this.setImp({ step:4, added: sel });
        if (this.mainRef.current) this.mainRef.current.scrollTop = 0;
      },
      goImpDeck: () => this.go('baralho', { deckId: imp.deck, search:'' }),
    };
  }

      memLegend: ml.map((l, i) => ({ l, c: mc[i], v: mem[5][i] })),
      memMonths: ['abr','mai','jun','jul','ago','set'].map((l, i) => ({ l, h: (mem[i].reduce((a, b) => a + b, 0) / mx * 100) + '%',

      leeches: s.leeches.map(l => ({ ...l, active: !l.susp, op: l.susp ? .55 : 1,

    };

  renderVals() {
    const s = this.state, P = this.props;
    const vp = P.viewport ?? 'auto';
    const showIntervals = P.showIntervals ?? true;
    const w = s.winW;
    const isMobile = vp === 'mobile' || (vp === 'auto' && w < 760);
    const isTablet = !isMobile && (vp === 'tablet' || (vp === 'auto' && w < 1100));
    const framed = vp === 'mobile' || vp === 'tablet';
    const sec = { inicio:'inicio', estudar:'estudar', sessao:'estudar', resultado:'estudar', baralhos:'baralhos', baralho:'baralhos', criar:'baralhos', importar:'baralhos', evolucao:'evolucao', ajustes:'ajustes', estados:'ajustes' }[s.screen];

      { id:'ajustes', n:'05', label:'Ajustes', tint:'#FAEBF0' },

      startAll: () => this.startSession(null),
    };
